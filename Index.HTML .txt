        function filterSignals(category) {
            currentCategory = category;
            
            // Update Tab Active Styles
            document.querySelectorAll('.tab-btn').forEach(btn => {
                if(btn.dataset.category === category) {
                    btn.className = "tab-btn px-4 py-2 rounded-xl text-sm font-semibold transition bg-blue-600 text-white shadow-lg shadow-blue-600/30";
                } else {
                    btn.className = "tab-btn px-4 py-2 rounded-xl text-sm font-semibold transition bg-slate-800 text-slate-400 hover:bg-slate-700";
                }
            });

            renderSignals();
        }

        function calculateLot() {
            const capital = parseFloat(document.getElementById('calcCapital').value) || 0;
            const riskPercent = parseFloat(document.getElementById('calcRisk').value) || 0;

            const riskAmount = capital * (riskPercent / 100);
            
            // Standard Lot Calculation for XAU/USD assuming 100 pips ($10) SL
            // $10 risk on 0.01 lot = $10 loss for a 100 pips move
            const safeLot = (riskAmount / 1000).toFixed(2);

            document.getElementById('riskAmountDisplay').innerText = `$${riskAmount.toFixed(2)}`;
            document.getElementById('safeLotDisplay').innerText = `${Math.max(safeLot, 0.01)} Lot`;
        }

        function refreshSignals() {
            const icon = document.getElementById('refreshIcon');
            icon.classList.add('fa-spin');
            setTimeout(() => {
                icon.classList.remove('fa-spin');
                updateTime();
            }, 800);
